cmake_minimum_required(VERSION 3.24)
project(ckgit_windows_vm_lifetime LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This opt-in HCS qualification requires native Windows and MSVC")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use a separate, task-owned development directory")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(ckgit-windows-vm-lifetime ../windows_vm_lifetime_probe.cpp)
target_compile_features(ckgit-windows-vm-lifetime PRIVATE cxx_std_20)
target_compile_options(ckgit-windows-vm-lifetime PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(ckgit-windows-vm-lifetime PRIVATE computecore ole32)
enable_testing()
add_test(NAME windows-vm-lifetime COMMAND ckgit-windows-vm-lifetime)
set_tests_properties(windows-vm-lifetime PROPERTIES TIMEOUT 150)
